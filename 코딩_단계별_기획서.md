### 1-1. SQLite 데이터베이스 초기화

**`data/database.py`**

```python
import sqlite3
import os
from config.settings import DB_PATH

def get_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_connection()
    cursor = conn.cursor()

    # 시장 데이터 (OHLCV)
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS market_data (
            id          INTEGER PRIMARY KEY AUTOINCREMENT,
            ticker      TEXT NOT NULL,
            date        TEXT NOT NULL,
            open        REAL,
            high        REAL,
            low         REAL,
            close       REAL,
            volume      INTEGER,
            UNIQUE(ticker, date)
        )
    ''')

    # 전략 파라미터
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS strategy_params (
            id          INTEGER PRIMARY KEY AUTOINCREMENT,
            ticker      TEXT NOT NULL,
            strategy    TEXT NOT NULL,
            params      TEXT,           -- JSON 문자열
            sharpe      REAL,
            mdd         REAL,
            win_rate    REAL,
            updated_at  TEXT DEFAULT (datetime('now','localtime')),
            UNIQUE(ticker, strategy)
        )
    ''')

    # 거래 기록
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS trades (
            id           INTEGER PRIMARY KEY AUTOINCREMENT,
            ticker       TEXT,
            side         TEXT,          -- BUY / SELL
            price        REAL,
            quantity     INTEGER,
            amount       REAL,
            strategy     TEXT,
            pnl          REAL,
            pnl_pct      REAL,
            executed_at  TEXT DEFAULT (datetime('now','localtime'))
        )
    ''')

    # 포트폴리오 스냅샷
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS portfolio_snapshots (
            id              INTEGER PRIMARY KEY AUTOINCREMENT,
            snapshot_date   TEXT,
            total_value     REAL,
            cash            REAL,
            positions       TEXT,       -- JSON 문자열
            daily_pnl       REAL,
            created_at      TEXT DEFAULT (datetime('now','localtime'))
        )
    ''')

    conn.commit()
    conn.close()
    print("[DB] 초기화 완료:", DB_PATH)

def insert_market_data(rows: list[dict]):
    conn = get_connection()
    cursor = conn.cursor()
    cursor.executemany('''
        INSERT OR REPLACE INTO market_data
        (ticker, date, open, high, low, close, volume)
        VALUES (:ticker, :date, :open, :high, :low, :close, :volume)
    ''', rows)
    conn.commit()
    conn.close()

def get_ohlcv(ticker: str, days: int = 200) -> list[dict]:
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute('''
        SELECT * FROM market_data
        WHERE ticker = ?
        ORDER BY date DESC
        LIMIT ?
    ''', (ticker, days))
    rows = [dict(r) for r in cursor.fetchall()]
    conn.close()
    return rows[::-1]   # 날짜 오름차순 반환

def insert_trade(trade: dict):
    conn = get_connection()
    conn.execute('''
        INSERT INTO trades (ticker, side, price, quantity, amount, strategy, pnl, pnl_pct)
        VALUES (:ticker, :side, :price, :quantity, :amount, :strategy, :pnl, :pnl_pct)
    ''', trade)
    conn.commit()
    conn.close()
```

### 1-2. 주식 데이터 수집기 (pykrx)

**`data/collector.py`**

```python
from pykrx import stock
import pandas as pd
from datetime import datetime, timedelta
from data.database import insert_market_data
import logging

logger = logging.getLogger(__name__)

def get_top_tickers(market: str = "KOSPI", top_n: int = 100) -> list[str]:
    """거래대금 상위 N개 종목 코드 반환"""
    today = datetime.now().strftime("%Y%m%d")
    df = stock.get_market_trading_value_by_ticker(today, market=market)
    df = df.sort_values("거래대금", ascending=False)
    return df.index.tolist()[:top_n]

def fetch_ohlcv(ticker: str, days: int = 365) -> pd.DataFrame:
    """단일 종목 OHLCV 수집"""
    end = datetime.now().strftime("%Y%m%d")
    start = (datetime.now() - timedelta(days=days)).strftime("%Y%m%d")
    df = stock.get_market_ohlcv_by_date(start, end, ticker)
    df.index = df.index.strftime("%Y-%m-%d")
    df.columns = ["open", "high", "low", "close", "volume", "change"]
    return df[["open", "high", "low", "close", "volume"]]

def fetch_and_store(ticker: str, days: int = 365):
    """수집 후 DB 저장"""
    try:
        df = fetch_ohlcv(ticker, days)
        rows = [
            {"ticker": ticker, "date": date,
             "open": row["open"], "high": row["high"],
             "low": row["low"],  "close": row["close"],
             "volume": row["volume"]}
            for date, row in df.iterrows()
        ]
        insert_market_data(rows)
        logger.info(f"[Collector] {ticker}: {len(rows)}건 저장 완료")
    except Exception as e:
        logger.error(f"[Collector] {ticker} 수집 실패: {e}")

def fetch_all_top(top_n: int = 100):
    """상위 종목 전체 수집"""
    tickers = get_top_tickers(top_n=top_n)
    for ticker in tickers:
        fetch_and_store(ticker)
    print(f"[Collector] 전체 {len(tickers)}종목 수집 완료")
```

### 1-3. 공시 수집기 (OpenDART)

**`data/dart_client.py`**

```python
import requests
import os
from datetime import datetime, timedelta

DART_API_KEY = os.getenv("DART_API_KEY")
BASE_URL = "https://opendart.fss.or.kr/api"

def get_recent_disclosures(corp_code: str = None, days: int = 1) -> list[dict]:
    """최근 N일 이내 공시 목록 조회"""
    end_date = datetime.now().strftime("%Y%m%d")
    start_date = (datetime.now() - timedelta(days=days)).strftime("%Y%m%d")

    params = {
        "crtfc_key": DART_API_KEY,
        "bgn_de": start_date,
        "end_de": end_date,
        "sort": "date",
        "sort_mth": "desc",
        "page_count": 40,
    }
    if corp_code:
        params["corp_code"] = corp_code

    response = requests.get(f"{BASE_URL}/list.json", params=params, timeout=10)
    data = response.json()

    if data.get("status") != "000":
        return []
    return data.get("list", [])

def is_material_disclosure(title: str) -> bool:
    """주요 공시 여부 판단 (매수 회피 필터)"""
    danger_keywords = ["유상증자", "전환사채", "횡령", "배임", "상장폐지", "자본잠식"]
    positive_keywords = ["자사주취득", "배당", "영업이익", "실적"]
    for kw in danger_keywords:
        if kw in title:
            return False   # 위험 공시 → 해당 종목 당일 매수 제외
    return True
```

```python
import requests
import os

TAVILY_API_KEY = os.getenv("TAVILY_API_KEY")

def search_stock_news(ticker_name: str, max_results: int = 5) -> list[dict]:
    """종목명 기준 최신 뉴스 검색 (Tavily — 월 1,000건 무료)"""
    headers = {"Content-Type": "application/json"}

        "query": f"{ticker_name} 주식 뉴스 오늘",
        "search_depth": "basic",
        "max_results": max_results,
        "include_answer": False,
    }
    response = requests.post(
        "https://api.tavily.com/search",
        json=payload,
        headers=headers,
        timeout=10
    )
    data = response.json()
    return data.get("results", [])

def get_market_sentiment_news(max_results: int = 10) -> list[dict]:
    """코스피 전반 시장 분위기 뉴스"""
```

### 2-1. 기술적 지표 계산기

**`strategy/indicators.py`**

```python
import pandas as pd
import numpy as np

def calc_rsi(close: pd.Series, period: int = 14) -> pd.Series:
    delta = close.diff()
    gain = delta.clip(lower=0).rolling(period).mean()
    loss = (-delta.clip(upper=0)).rolling(period).mean()
    rs = gain / loss
    return 100 - (100 / (1 + rs))

def calc_bollinger_bands(close: pd.Series, period: int = 20, std: float = 2.0):
    mid = close.rolling(period).mean()
    sigma = close.rolling(period).std()
    return mid + std * sigma, mid, mid - std * sigma  # upper, mid, lower

def calc_macd(close: pd.Series, fast=12, slow=26, signal=9):
    ema_fast = close.ewm(span=fast).mean()
    ema_slow = close.ewm(span=slow).mean()
    macd = ema_fast - ema_slow
    signal_line = macd.ewm(span=signal).mean()
    histogram = macd - signal_line
    return macd, signal_line, histogram

def calc_atr(high: pd.Series, low: pd.Series, close: pd.Series, period: int = 14) -> pd.Series:
    tr = pd.concat([
        high - low,
        (high - close.shift()).abs(),
        (low - close.shift()).abs()
    ], axis=1).max(axis=1)
    return tr.rolling(period).mean()

def calc_vwap(high: pd.Series, low: pd.Series, close: pd.Series, volume: pd.Series) -> pd.Series:
    typical_price = (high + low + close) / 3
    return (typical_price * volume).cumsum() / volume.cumsum()

def calc_sma(close: pd.Series, period: int) -> pd.Series:
    return close.rolling(period).mean()

def add_all_indicators(df: pd.DataFrame) -> pd.DataFrame:
    """DataFrame에 모든 지표 컬럼 추가"""
    df = df.copy()
    df["rsi_14"]          = calc_rsi(df["close"], 14)
    df["rsi_21"]          = calc_rsi(df["close"], 21)
    df["bb_upper"], df["bb_mid"], df["bb_lower"] = calc_bollinger_bands(df["close"])
    df["macd"], df["macd_signal"], df["macd_hist"] = calc_macd(df["close"])
    df["atr_14"]          = calc_atr(df["high"], df["low"], df["close"], 14)
    df["atr_pct"]         = df["atr_14"] / df["close"] * 100
    df["sma_20"]          = calc_sma(df["close"], 20)
    df["sma_60"]          = calc_sma(df["close"], 60)
    df["sma_120"]         = calc_sma(df["close"], 120)
    df["vwap"]            = calc_vwap(df["high"], df["low"], df["close"], df["volume"])
    return df
```

### 2-2. 래리 윌리엄스 변동성 돌파 전략

**`strategy/larry_williams.py`**

```python
import pandas as pd
from strategy.indicators import calc_atr
from data.collector import fetch_ohlcv
from datetime import datetime
import logging

logger = logging.getLogger(__name__)

def calc_trigger_price(
    open_price: float,
    high_14min: float,
    low_14min: float,
    k: float = 0.5
) -> float:
    """장 개시 후 14분 레인지로 매수 트리거가 계산"""
    range_14 = high_14min - low_14min
    return open_price + (range_14 * k)

def calc_position_size(total_capital: float, atr_pct: float) -> float:
    """ATR 기반 동적 포지션 사이징"""
    if atr_pct >= 3.0:
        ratio = 0.05     # 고변동성: 자본의 5%
    elif atr_pct >= 1.5:
        ratio = 0.10     # 중변동성: 자본의 10%
    elif atr_pct >= 0.5:
        ratio = 0.20     # 저변동성: 자본의 20%
    else:
        ratio = 0.0      # 매수 보류
    return total_capital * ratio

def get_candidate_tickers(top_n: int = 10) -> list[dict]:
    """변동성 돌파 전략 대상 종목 스크리닝"""
    from data.collector import get_top_tickers
    from data.database import get_ohlcv
    import pandas as pd

    tickers = get_top_tickers(top_n=50)
    candidates = []

    for ticker in tickers:
        rows = get_ohlcv(ticker, days=20)
        if len(rows) < 15:
            continue

        df = pd.DataFrame(rows)
        df["close"] = df["close"].astype(float)
        df["high"]  = df["high"].astype(float)
        df["low"]   = df["low"].astype(float)

        atr = calc_atr(df["high"], df["low"], df["close"], 14).iloc[-1]
        last_close = df["close"].iloc[-1]
        atr_pct = atr / last_close * 100

        # ATR 0.5% 미만 제외 (유동성 부족)
        if atr_pct < 0.5:
            continue

        candidates.append({
            "ticker": ticker,
            "last_close": last_close,
            "atr_pct": round(atr_pct, 2),
        })

    return sorted(candidates, key=lambda x: x["atr_pct"])[:top_n]

def check_breakout(
    ticker: str,
    current_price: float,
    trigger_price: float,
    current_volume: int,
    avg_volume: float,
    volume_ratio_min: float = 1.3
) -> bool:
    """현재가가 트리거가를 돌파했는지 + 거래량 필터 확인"""
    price_ok  = current_price >= trigger_price
    volume_ok = current_volume >= avg_volume * volume_ratio_min
    return price_ok and volume_ok

class LarryWilliamsStrategy:
    """래리 윌리엄스 변동성 돌파 전략 상태 관리"""

    def __init__(self, total_capital: float, k: float = 0.5):
        self.total_capital = total_capital
        self.k = k
        self.triggers: dict[str, float] = {}   # ticker → 트리거가
        self.positions: dict[str, dict] = {}   # ticker → 보유 정보

    def set_trigger(self, ticker: str, open_p: float, high_14: float, low_14: float):
        self.triggers[ticker] = calc_trigger_price(open_p, high_14, low_14, self.k)
        logger.info(f"[LW] {ticker} 트리거가: {self.triggers[ticker]:,.0f}원")

    def should_buy(self, ticker: str, current_price: float,
                   current_volume: int, avg_volume: float) -> bool:
        trigger = self.triggers.get(ticker)
        if trigger is None:
            return False
        return check_breakout(ticker, current_price, trigger, current_volume, avg_volume)

    def get_buy_amount(self, atr_pct: float) -> float:
        return calc_position_size(self.total_capital, atr_pct)

    def should_sell(self, ticker: str, current_price: float,
                    buy_price: float, take_profit=0.03, stop_loss=-0.02) -> str | None:
        if ticker not in self.positions:
            return None
        pnl_pct = (current_price - buy_price) / buy_price
        if pnl_pct >= take_profit:
            return "TAKE_PROFIT"
        if pnl_pct <= stop_loss:
            return "STOP_LOSS"
        return None
```

```python
import pandas as pd
import numpy as np
import json
from itertools import product
from strategy.indicators import add_all_indicators, calc_rsi, calc_bollinger_bands, calc_sma
from data.database import get_ohlcv, get_connection

# ─────────────────────────────────────────
# 전략 정의 (총 44개 — 여기서는 핵심 10개 예시)
# ─────────────────────────────────────────
STRATEGIES = {
    "RSI_MeanReversion": {
        "params": {
            "period":    [7, 10, 14, 21],
            "oversold":  [20, 25, 30, 35],
            "overbought":[65, 70, 75, 80],
        },
        "signal_fn": "rsi_signal",
    },
    "BB_MeanReversion": {
        "params": {

        },
        "signal_fn": "bb_signal",
    },
    "SMA_Cross": {
        "params": {
            "fast": [5, 10, 20],
            "slow": [60, 100, 120, 200],
        },
        "signal_fn": "sma_cross_signal",
    },
    # ... 나머지 전략 추가
}

def rsi_signal(df: pd.DataFrame, period=14, oversold=30, overbought=70) -> pd.Series:
    rsi = calc_rsi(df["close"], period)
    signal = pd.Series(0, index=df.index)
    signal[rsi < oversold]  =  1   # 매수
    signal[rsi > overbought] = -1  # 매도
    return signal

def bb_signal(df: pd.DataFrame, period=20, std=2.0) -> pd.Series:
    upper, mid, lower = calc_bollinger_bands(df["close"], period, std)
    signal = pd.Series(0, index=df.index)
    signal[df["close"] <= lower] =  1
    signal[df["close"] >= upper] = -1
    return signal

def sma_cross_signal(df: pd.DataFrame, fast=20, slow=100) -> pd.Series:
    sma_fast = calc_sma(df["close"], fast)
    sma_slow = calc_sma(df["close"], slow)
    signal = pd.Series(0, index=df.index)
    signal[(sma_fast > sma_slow) & (sma_fast.shift() <= sma_slow.shift())] =  1
    signal[(sma_fast < sma_slow) & (sma_fast.shift() >= sma_slow.shift())] = -1
    return signal

def backtest(df: pd.DataFrame, signal: pd.Series, initial=10_000_000) -> dict:
    """간단한 벡터화 백테스트"""
    position = 0
    cash = initial
    portfolio = initial
    buy_price = 0
    trades, wins = 0, 0
    peak = initial
    max_drawdown = 0

    for i in range(1, len(df)):
        sig = signal.iloc[i]
        price = df["close"].iloc[i]

        if sig == 1 and position == 0:
            position = cash // price
            cash -= position * price
            buy_price = price

        elif sig == -1 and position > 0:
            cash += position * price
            trades += 1
            if price > buy_price:
                wins += 1
            position = 0

        portfolio = cash + position * price
        peak = max(peak, portfolio)
        dd = (portfolio - peak) / peak
        max_drawdown = min(max_drawdown, dd)

    total_return = (portfolio - initial) / initial
    win_rate = wins / trades if trades > 0 else 0

    # 샤프 지수 (단순화)
    daily_returns = df["close"].pct_change().dropna()
    sharpe = (daily_returns.mean() / daily_returns.std()) * (252 ** 0.5) if daily_returns.std() > 0 else 0

    return {
        "total_return": round(total_return, 4),
        "mdd":          round(max_drawdown, 4),
        "win_rate":     round(win_rate, 4),
        "trades":       trades,
        "sharpe":       round(sharpe, 4),
    }

def grid_search(ticker: str, strategy_name: str) -> dict | None:
    """단일 종목 × 단일 전략 그리드 서치"""
    rows = get_ohlcv(ticker, days=365 * 3)   # 3년치
    if len(rows) < 200:
        return None

    df = pd.DataFrame(rows)
    df["close"] = df["close"].astype(float)
    df["high"]  = df["high"].astype(float)
    df["low"]   = df["low"].astype(float)

    strategy = STRATEGIES.get(strategy_name)
    if not strategy:
        return None

    best = {"sharpe": -999}
    param_grid = strategy["params"]
    keys = list(param_grid.keys())
    values = list(param_grid.values())

    for combo in product(*values):
        params = dict(zip(keys, combo))
        fn = globals()[strategy["signal_fn"]]
        try:
            signal = fn(df, **params)
            result = backtest(df, signal)
        except Exception:
            continue

        # 샤프 지수 우선, MDD 조건 충족 시만 후보
        if result["sharpe"] > best["sharpe"] and result["mdd"] > -0.20 and result["win_rate"] >= 0.45:
            best = {**result, "params": params, "strategy": strategy_name, "ticker": ticker}

    return best if "params" in best else None

def run_full_optimization(tickers: list[str]):
    """전 종목 × 전 전략 최적화 실행 (월 1회 야간 배치)"""
    conn = get_connection()
    saved = 0
    for ticker in tickers:
        for strategy_name in STRATEGIES:
            result = grid_search(ticker, strategy_name)
            if result and result.get("sharpe", 0) >= 1.0:
                conn.execute('''
                    INSERT OR REPLACE INTO strategy_params
                    (ticker, strategy, params, sharpe, mdd, win_rate)
                    VALUES (?, ?, ?, ?, ?, ?)
                ''', (
                    ticker, strategy_name,
                    json.dumps(result["params"]),
                    result["sharpe"],
                    result["mdd"],
                    result["win_rate"],
                ))
                saved += 1
    conn.commit()
    conn.close()
    logger.info(f"[Optimizer] 최적화 완료: {saved}개 전략-종목 조합 저장")
```

```python
import pandas as pd
from data.database import get_ohlcv, get_connection
from strategy.indicators import add_all_indicators
import json

def score_ticker(ticker: str) -> float:
    """
    멀티팩터 스코어링 (0~100점)
    - 가격 변동성 (30점)
    - 수급 (30점)
    - 기술적 신호 (40점)
    """
    rows = get_ohlcv(ticker, days=30)
    if len(rows) < 10:
        return 0.0

    df = pd.DataFrame(rows)
    df = df.apply(pd.to_numeric, errors="coerce")
    df = add_all_indicators(df)
    last = df.iloc[-1]

    score = 0.0

    # ① 가격 변동성 점수 (ATR_pct 0.5~2.5% 구간이 이상적)
    atr_pct = last.get("atr_pct", 0)
    if 0.5 <= atr_pct <= 2.5:
        score += 30
    elif atr_pct > 2.5:
        score += 15   # 과고변동 — 절반만

    # ② 수급 점수 (거래량이 5일 평균의 1.5배 이상)
    vol_ratio = df["volume"].iloc[-1] / df["volume"].iloc[-6:-1].mean()
    if vol_ratio >= 2.0:
        score += 30
    elif vol_ratio >= 1.5:
        score += 20
    elif vol_ratio >= 1.0:
        score += 10

    # ③ 기술적 신호 점수
    rsi = last.get("rsi_14", 50)
    close = last.get("close", 0)
    bb_lower = last.get("bb_lower", 0)
    sma20 = last.get("sma_20", 0)
    macd = last.get("macd", 0)
    macd_sig = last.get("macd_signal", 0)

    if rsi < 35:             score += 15   # 과매도
    if close <= bb_lower:    score += 15   # 볼린저 하단
    if macd > macd_sig:      score += 10   # MACD 골든크로스

    return min(score, 100.0)

def get_buy_candidates(min_score: float = 70.0, top_n: int = 5) -> list[dict]:
    """스코어 기준 매수 후보 종목 반환"""
    from data.collector import get_top_tickers
    tickers = get_top_tickers(top_n=50)

    results = []
    for ticker in tickers:
        sc = score_ticker(ticker)
        if sc >= min_score:
            results.append({"ticker": ticker, "score": sc})

    results.sort(key=lambda x: x["score"], reverse=True)
    return results[:top_n]
```

```python
import requests
import os
import json
from datetime import datetime, timedelta
import logging

logger = logging.getLogger(__name__)

class KISClient:
    def __init__(self):
        self.app_key    = os.getenv("KIS_APP_KEY")
        self.app_secret = os.getenv("KIS_APP_SECRET")
        self.account_no = os.getenv("KIS_ACCOUNT_NO")
        self.is_paper   = os.getenv("KIS_MODE", "paper") == "paper"

        self.base_url = (
            "https://openapivts.koreainvestment.com:29443"   # 모의
            if self.is_paper else
            "https://openapi.koreainvestment.com:9443"        # 실전
        )
        self._token: str | None = None
        self._token_expires: datetime | None = None

    def _get_token(self) -> str:
        """OAuth2 액세스 토큰 발급 (유효기간 24시간)"""
        if self._token and self._token_expires and datetime.now() < self._token_expires:
            return self._token

        url = f"{self.base_url}/oauth2/tokenP"
        payload = {
            "grant_type": "client_credentials",
            "appkey": self.app_key,
            "appsecret": self.app_secret,
        }
        resp = requests.post(url, json=payload, timeout=10)
        resp.raise_for_status()
        data = resp.json()
        self._token = data["access_token"]
        self._token_expires = datetime.now() + timedelta(hours=23)
        return self._token

    def _headers(self, tr_id: str) -> dict:
        return {
            "Content-Type": "application/json",
            "authorization": f"Bearer {self._get_token()}",
            "appkey": self.app_key,
            "appsecret": self.app_secret,
            "tr_id": tr_id,
        }

    def get_current_price(self, ticker: str) -> float:
        """현재가 조회"""
        url = f"{self.base_url}/uapi/domestic-stock/v1/quotations/inquire-price"
        params = {"FID_COND_MRKT_DIV_CODE": "J", "FID_INPUT_ISCD": ticker}
        tr_id = "FHKST01010100"
        resp = requests.get(url, headers=self._headers(tr_id), params=params, timeout=10)
        data = resp.json()
        return float(data["output"]["stck_prpr"])

    def place_order(self, ticker: str, side: str, quantity: int, price: int = 0) -> dict:
        """
        주문 실행
        side: "BUY" or "SELL"
        price: 0이면 시장가
        """
        if os.getenv("DRY_RUN", "True") == "True":
            logger.info(f"[DRY_RUN] {side} {ticker} {quantity}주 @ {price or '시장가'}")
            return {"status": "dry_run", "ticker": ticker, "side": side, "quantity": quantity}

        url = f"{self.base_url}/uapi/domestic-stock/v1/trading/order-cash"
        tr_id = "VTTC0802U" if self.is_paper else "TTTC0802U"   # 매수
        if side == "SELL":
            tr_id = "VTTC0801U" if self.is_paper else "TTTC0801U"

        order_dv = "01" if price == 0 else "00"  # 01=시장가, 00=지정가
        payload = {
            "CANO": self.account_no[:8],
            "ACNT_PRDT_CD": self.account_no[-2:],
            "PDNO": ticker,
            "ORD_DVSN": order_dv,
            "ORD_QTY": str(quantity),
            "ORD_UNPR": str(price),
        }
        resp = requests.post(url, headers=self._headers(tr_id), json=payload, timeout=10)
        resp.raise_for_status()
        result = resp.json()
        logger.info(f"[KIS] {side} {ticker} {quantity}주 체결: {result}")
        return result

    def get_balance(self) -> dict:
        """잔고 및 보유 종목 조회"""
        url = f"{self.base_url}/uapi/domestic-stock/v1/trading/inquire-balance"
        tr_id = "VTTC8434R" if self.is_paper else "TTTC8434R"
        params = {
            "CANO": self.account_no[:8],
            "ACNT_PRDT_CD": self.account_no[-2:],
            "AFHR_FLPR_YN": "N", "OFL_YN": "N",
            "INQR_DVSN": "01", "UNPR_DVSN": "01",
            "FUND_STTL_ICLD_YN": "N", "FNCG_AMT_AUTO_RDPT_YN": "N",
            "PRCS_DVSN": "01", "CTX_AREA_FK100": "", "CTX_AREA_NK100": "",
        }
        resp = requests.get(url, headers=self._headers(tr_id), params=params, timeout=10)
```

```python
import json
from broker.kis_api import KISClient
from data.database import insert_trade, get_connection
from risk.position_sizer import calc_position_size
from notify.telegram_bot import send_message
import logging
import os

logger = logging.getLogger(__name__)
MAX_SLOTS = int(os.getenv("MAX_SLOTS", 5))

class OrderManager:
    def __init__(self):
        self.client = KISClient()
        self.positions: dict[str, dict] = {}   # ticker → {price, quantity, strategy}

    def available_slots(self) -> int:
        return MAX_SLOTS - len(self.positions)

    def buy(self, ticker: str, strategy: str, atr_pct: float) -> bool:
        if self.available_slots() <= 0:
            logger.warning("[OM] 빈 슬롯 없음 — 매수 건너뜀")
            return False

        balance = self.client.get_balance()
        cash = float(balance["output2"][0]["dnca_tot_amt"])
        amount = calc_position_size(cash, atr_pct)

        if amount < 10_000:
            logger.warning(f"[OM] 매수금액 너무 작음: {amount:,.0f}원")
            return False

        current_price = self.client.get_current_price(ticker)
        quantity = int(amount // current_price)

        if quantity < 1:
            return False

        result = self.client.place_order(ticker, "BUY", quantity)
        if result.get("status") in ("dry_run",) or result.get("rt_cd") == "0":
            self.positions[ticker] = {
                "buy_price": current_price,
                "quantity": quantity,
                "strategy": strategy,
                "amount": quantity * current_price,
            }
            insert_trade({
                "ticker": ticker, "side": "BUY",
                "price": current_price, "quantity": quantity,
                "amount": quantity * current_price,
                "strategy": strategy, "pnl": 0, "pnl_pct": 0,
            })
            send_message(f"✅ 매수 체결\n종목: {ticker}\n수량: {quantity}주\n단가: {current_price:,.0f}원\n전략: {strategy}")
            return True
        return False

    def sell(self, ticker: str, reason: str = "") -> bool:
        if ticker not in self.positions:
            return False

        pos = self.positions[ticker]
        current_price = self.client.get_current_price(ticker)
        result = self.client.place_order(ticker, "SELL", pos["quantity"])

        pnl = (current_price - pos["buy_price"]) * pos["quantity"]
        pnl_pct = (current_price - pos["buy_price"]) / pos["buy_price"] * 100

        insert_trade({
            "ticker": ticker, "side": "SELL",
            "price": current_price, "quantity": pos["quantity"],
            "amount": current_price * pos["quantity"],
            "strategy": pos["strategy"],
            "pnl": round(pnl, 0), "pnl_pct": round(pnl_pct, 2),
        })

        emoji = "💰" if pnl > 0 else "🔴"
        send_message(
            f"{emoji} 매도 체결 ({reason})\n"
            f"종목: {ticker}\n수량: {pos['quantity']}주\n"
            f"단가: {current_price:,.0f}원\n"
            f"손익: {pnl:+,.0f}원 ({pnl_pct:+.2f}%)"
        )
        del self.positions[ticker]
        return True

    def sell_all(self, reason: str = "장마감_일괄청산"):
        for ticker in list(self.positions.keys()):
            self.sell(ticker, reason)
```

### 6-1. Circuit Breaker

**`risk/circuit_breaker.py`**

```python
import os
import logging
from notify.telegram_bot import send_message

logger = logging.getLogger(__name__)
DAILY_LOSS_LIMIT = float(os.getenv("DAILY_LOSS_LIMIT", -0.03))

class CircuitBreaker:
    def __init__(self):
        self._triggered = False
        self._start_value: float | None = None

    def reset(self):
        self._triggered = False
        self._start_value = None

    def set_start(self, value: float):
        self._start_value = value

    def check(self, kis_client) -> bool:
        if self._triggered:
            return True
        if self._start_value is None:
            return False

        balance = kis_client.get_balance()
        current = float(balance["output2"][0]["tot_evlu_amt"])
        loss_pct = (current - self._start_value) / self._start_value

        if loss_pct <= DAILY_LOSS_LIMIT:
            self._triggered = True
            msg = (
                f"🚨 Circuit Breaker 발동!\n"
                f"일일 손실 {loss_pct*100:.2f}% (한도: {DAILY_LOSS_LIMIT*100:.0f}%)\n"
                f"당일 전략 매매 중단"
            )
            send_message(msg)
            logger.warning(msg)
            return True
        return False

    def is_triggered(self) -> bool:
        return self._triggered
```

```python
import os

class HaikuClient:
    def __init__(self):

        response = self.client.messages.create(**kwargs)
        return response.content[0].text
```

```python
    # Claude Haiku AI 분석 (1일 1회)
    ai_comment = analyze_today_trades()
    lines.append(f"🤖 AI 분석: {ai_comment}")

    return "\n".join(lines)
```

## Phase 9 — 멀티 에이전트 자기학습 루프

> **목표**: 매일 장 마감 후 거래 결과 → Haiku 분석 → 파라미터 자동 조정  
> **파일**: `learning/feedback_loop.py`, `learning/param_updater.py`

### 9-1. 피드백 루프

**`learning/feedback_loop.py`**

```python
import json
from data.database import get_connection
from ai.analyzer import analyze_today_trades
from learning.param_updater import update_strategy_weights
from notify.telegram_bot import send_message
from datetime import datetime
import logging

logger = logging.getLogger(__name__)

def run_daily_feedback():
    """장 마감 후 1회 실행 — 15:35 크론잡"""
    logger.info("[Feedback] 자기학습 피드백 루프 시작")

    conn = get_connection()

    # 오늘 전략별 성과 집계
    cursor = conn.cursor()
    cursor.execute("""
        SELECT strategy,
               COUNT(*) as cnt,
               SUM(CASE WHEN pnl > 0 THEN 1 ELSE 0 END) as wins,
               AVG(pnl_pct) as avg_pnl_pct
        FROM trades
        WHERE date(executed_at) = date('now', 'localtime')
          AND side = 'SELL'
        GROUP BY strategy
    """)
    results = [dict(r) for r in cursor.fetchall()]
    conn.close()

    if not results:
        return

    # 손실 전략 가중치 하향, 수익 전략 상향
    for r in results:
        win_rate = r["wins"] / r["cnt"] if r["cnt"] > 0 else 0
        update_strategy_weights(r["strategy"], win_rate, r["avg_pnl_pct"])

    # memory.md 업데이트
    _update_memory(results)
    logger.info("[Feedback] 파라미터 업데이트 완료")

def _update_memory(results: list[dict]):
    """오늘 학습 내용을 memory.md에 추가"""
    today = datetime.now().strftime("%Y-%m-%d")
    new_entry = f"\n## {today} 학습\n"
    for r in results:
        win_rate = r["wins"] / r["cnt"] * 100 if r["cnt"] > 0 else 0
        new_entry += f"- {r['strategy']}: 승률 {win_rate:.0f}%, 평균 수익률 {r['avg_pnl_pct']:.2f}%\n"

    with open("agent/memory.md", "a", encoding="utf-8") as f:
        f.write(new_entry)
```

### 9-2. 파라미터 자동 조정

**`learning/param_updater.py`**

```python
import json
from data.database import get_connection
import logging

logger = logging.getLogger(__name__)

def update_strategy_weights(strategy: str, win_rate: float, avg_pnl_pct: float):
    """
    승률·손익률 기반 전략 파라미터 우선순위 조정
    - 승률 < 40% → 해당 전략 비중 축소 (score_threshold 상향)
    - 승률 > 65% → 해당 전략 비중 확대 (score_threshold 하향)
    """
    conn = get_connection()
    cursor = conn.cursor()

    cursor.execute("SELECT params FROM strategy_params WHERE strategy = ? LIMIT 1", (strategy,))
    row = cursor.fetchone()
    if not row:
        conn.close()
        return

    params = json.loads(row["params"]) if row["params"] else {}

    # 승률 기반 가중치 조정
    current_threshold = params.get("score_threshold", 70)
    if win_rate < 0.40:
        params["score_threshold"] = min(current_threshold + 5, 90)
        logger.info(f"[Updater] {strategy} 스코어 임계값 상향: {current_threshold} → {params['score_threshold']}")
    elif win_rate > 0.65:
        params["score_threshold"] = max(current_threshold - 3, 60)
        logger.info(f"[Updater] {strategy} 스코어 임계값 하향: {current_threshold} → {params['score_threshold']}")

    cursor.execute(
        "UPDATE strategy_params SET params = ? WHERE strategy = ?",
        (json.dumps(params), strategy)
    )
    conn.commit()
    conn.close()
```
